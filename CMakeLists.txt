cmake_minimum_required(VERSION 3.22)
project(hyprstack VERSION 1.3.0 LANGUAGES CXX)
include(GNUInstallDirs)
set(CMAKE_INSTALL_LIBDIR "${CMAKE_INSTALL_DATADIR}/hyprstack/lib" CACHE PATH "Private runtime libraries" FORCE)

set(VERSION "${PROJECT_VERSION}")
set(GIT_REVISION "release")
set(DISTRIBUTOR "Hyprstack")
set(INSTALL_LIBDIR "${CMAKE_INSTALL_DATADIR}/hyprstack/lib" CACHE STRING "Helper directory" FORCE)
set(INSTALL_QMLDIR "${CMAKE_INSTALL_DATADIR}/hyprstack/qml" CACHE STRING "Private QML modules" FORCE)
set(INSTALL_QSCONFDIR "${CMAKE_INSTALL_DATADIR}/hyprstack/shell" CACHE STRING "Shell directory" FORCE)
option(HYPRSTACK_BUILD_NATIVE "Build matching compositor extensions" ON)

find_package(Qt6 6.9 REQUIRED COMPONENTS Core Qml Quick ShaderTools)
set(M3SHAPES_INSTALL_DEVELOPMENT OFF CACHE BOOL "Install development files" FORCE)
add_subdirectory(third_party/m3shapes)
add_subdirectory(shell)

if(HYPRSTACK_BUILD_NATIVE)
    execute_process(COMMAND Hyprland --version-json OUTPUT_VARIABLE hypr_version
        RESULT_VARIABLE version_result OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(NOT version_result EQUAL 0)
        message(FATAL_ERROR "Install Hyprland 0.57.0 and matching development headers first")
    endif()
    string(JSON hypr_tag ERROR_VARIABLE version_error GET "${hypr_version}" version)
    if(version_error OR NOT hypr_tag STREQUAL "0.57.0")
        message(FATAL_ERROR "This native extension supports Hyprland 0.57.0. See INSTALL.md")
    endif()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(Native REQUIRED IMPORTED_TARGET hyprland pixman-1 libdrm libinput libudev wayland-server xkbcommon pangocairo)
    find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
    string(JSON hypr_abi ERROR_VARIABLE abi_error GET "${hypr_version}" abiHash)
    if(abi_error)
        message(FATAL_ERROR "Hyprland must report its native plugin ABI")
    endif()
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/distribution/check_headers.py" "${hypr_abi}"
        RESULT_VARIABLE headers_result OUTPUT_VARIABLE headers_output ERROR_VARIABLE headers_error)
    if(NOT headers_result EQUAL 0)
        message(FATAL_ERROR "${headers_error}")
    endif()
    message(STATUS "${headers_output}")
    add_library(hyprstack-bars MODULE native/hyprbars/main.cpp native/hyprbars/barDeco.cpp native/hyprbars/BarPassElement.cpp)
    set_target_properties(hyprstack-bars PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON PREFIX "" )
    target_link_libraries(hyprstack-bars PRIVATE PkgConfig::Native)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(hyprstack-bars PRIVATE -fno-gnu-unique)
    endif()
    target_compile_options(hyprstack-bars PRIVATE -Wno-narrowing)
    install(TARGETS hyprstack-bars LIBRARY DESTINATION "${CMAKE_INSTALL_DATADIR}/hyprstack/native")
    file(WRITE "${CMAKE_BINARY_DIR}/native-version.json" "${hypr_version}\n")
    install(FILES "${CMAKE_BINARY_DIR}/native-version.json" DESTINATION "${CMAKE_INSTALL_DATADIR}/hyprstack/native")
endif()

install(DIRECTORY distribution/defaults/ DESTINATION "${CMAKE_INSTALL_DATADIR}/hyprstack/defaults")
install(PROGRAMS distribution/bin/hyprstack distribution/bin/hyprstack-shell distribution/bin/hyprstack-session distribution/bin/hyprstack-session-runner distribution/bin/hyprstack-doctor distribution/bin/hyprstack-vpn DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES distribution/systemd/user/hyprstack-session.service distribution/systemd/user/hyprstack-vpn.service DESTINATION "${CMAKE_INSTALL_DATADIR}/hyprstack/systemd")
install(FILES shell/LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/hyprstack" RENAME LICENSE)
install(FILES THIRD_PARTY.md DESTINATION "${CMAKE_INSTALL_DATADIR}/hyprstack")
install(FILES native/hyprbars/LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/hyprstack/licenses" RENAME hyprbars-BSD-3-Clause)
install(FILES third_party/m3shapes/LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/hyprstack/licenses" RENAME M3Shapes-Apache-2.0)
install(FILES shell/assets/logo.svg DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps" RENAME hyprstack.svg)
